& {
    $ErrorActionPreference = 'Stop'
    $repo = 'C:\Users\anton\OneDrive\Documentos\GitHub\app_cane_prep'
    $work = 'C:\Users\anton\AppData\Local\Temp\cane-real-9c219c00d82741c7ae168a7dda6d5cff'
    $realRef = 'lzzyuxploolyqgitjtdu'
    $expected = [ordered]@{
        '20261009212148_supabase_cane.sql'               = 'dc7e92cdfb3887af1c349f9fed15d9e5f3716f881577b188e2a8fa43480c9831'
        '20261013034510_curacion_banco.sql'              = 'c9eefbb20ed7046a832319b2de4e7e9d304a28076024990a0eb3c7747f4929ea'
        '20261013044404_question_assets.sql'             = 'ab262f6bf57ec53c4b697d179ebbf0d2bfde6cdf011d183cd3ff12aae86e1a51'
        '20261014185955_revoke_service_role_secrets.sql' = '95524f4ae2eb7b935c171ae9c7e79eae0ad192ed35f4844df5211fdd2efb9bd4'
    }
    $names = @($expected.Keys)

    function Get-LfHash {
        param([string]$Path)
        $bytes = [System.IO.File]::ReadAllBytes($Path)
        $list = New-Object 'System.Collections.Generic.List[byte]'
        for ($i = 0; $i -lt $bytes.Length; $i++) {
            if ($bytes[$i] -eq 13 -and ($i + 1) -lt $bytes.Length -and $bytes[$i + 1] -eq 10) { continue }
            $list.Add($bytes[$i])
        }
        $sha = [System.Security.Cryptography.SHA256]::Create()
        return ([System.BitConverter]::ToString($sha.ComputeHash($list.ToArray())).Replace('-', '')).ToLower()
    }

    function Get-RawHash {
        param([string]$Path)
        return (Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLower()
    }

    $srcDir = Join-Path $repo 'supabase\migrations'
    $dstDir = Join-Path $work 'supabase\migrations'
    $linkFile = Join-Path $work 'supabase\.temp\project-ref'

    if (-not (Test-Path -LiteralPath $srcDir -PathType Container)) { throw 'No existe supabase\migrations en el repositorio.' }
    if (-not (Test-Path -LiteralPath $work -PathType Container)) { throw 'No existe la carpeta vinculada al proyecto real.' }
    if (-not (Test-Path -LiteralPath $dstDir -PathType Container)) { throw 'Falta supabase\migrations en la carpeta vinculada.' }
    if (-not (Test-Path -LiteralPath $linkFile -PathType Leaf)) { throw 'Falta supabase\.temp\project-ref en la carpeta vinculada.' }
    if (-not (Test-Path -LiteralPath (Join-Path $work 'supabase\config.toml') -PathType Leaf)) { throw 'Falta supabase\config.toml en la carpeta vinculada.' }
    if ((Get-Content -LiteralPath $linkFile -Raw).Trim() -cne $realRef) { throw 'El project-ref de la carpeta no es el del proyecto real: no se usa.' }

    $repoNames = @()
    foreach ($f in Get-ChildItem -LiteralPath $srcDir -File -Force) {
        if ($f.Name -ne '.gitkeep') { $repoNames += $f.Name }
    }
    $dstNames = @()
    foreach ($f in Get-ChildItem -LiteralPath $dstDir -File -Force) { $dstNames += $f.Name }
    foreach ($n in $names) {
        if ($repoNames -notcontains $n) { throw ('Falta en el repositorio: ' + $n) }
        if ($dstNames -notcontains $n) { throw ('Falta en la carpeta vinculada: ' + $n) }
    }
    foreach ($n in $repoNames) {
        if ($names -notcontains $n) { throw ('Archivo inesperado en el repositorio: ' + $n) }
    }
    foreach ($n in $dstNames) {
        if ($names -notcontains $n) { throw ('Archivo inesperado en la carpeta vinculada: ' + $n) }
    }
    foreach ($n in $names) {
        $rawRepo = Get-RawHash (Join-Path $srcDir $n)
        $rawDst = Get-RawHash (Join-Path $dstDir $n)
        if ($rawRepo -cne $rawDst) { throw ('La copia no coincide con el repositorio (raw): ' + $n) }
        if ((Get-LfHash (Join-Path $dstDir $n)) -cne $expected[$n]) { throw ('El hash LF de la copia no coincide con el esperado: ' + $n) }
    }
    Write-Output ('Carpeta verificada: ' + $work)
    Write-Output ('project-ref: ' + $realRef + ' | migraciones: ' + $names.Count + ' | raw=repo y LF=esperado en todas')
    Write-Output 'Ejecutando db push --dry-run (solo lectura; no aplica nada)...'

    npx --no-install supabase db push --dry-run --skip-vault --linked --workdir $work
    if ($LASTEXITCODE -ne 0) { throw ('db push --dry-run termino con codigo ' + $LASTEXITCODE) }
}
